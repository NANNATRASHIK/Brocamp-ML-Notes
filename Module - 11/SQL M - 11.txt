-- INSERT
INSERT INTO students (id, name, age, course, marks)
VALUES (10, 'Rashid', 19, 'Data Science', 87);

-- SELECT everything
SELECT * FROM students;

-- SELECT specific columns
SELECT name, marks
FROM students;

-- SELECT with filtering
SELECT * FROM students
WHERE marks > 80;

-- UPDATE
UPDATE students
SET marks = 90
WHERE id = 10;

-- DELETE
DELETE FROM students
WHERE id = 10;

-- INNER JOIN
-- Get students who have matching marks records
SELECT students.name, marks.marks
FROM students
INNER JOIN marks
ON students.id = marks.student_id;

-- Another INNER JOIN example
SELECT students.name, students.course, marks.marks
FROM students
INNER JOIN marks
ON students.id = marks.student_id;

-- LEFT JOIN
-- Get all students, even if they don't have marks
SELECT students.name, marks.marks
FROM students
LEFT JOIN marks
ON students.id = marks.student_id;

-- Another LEFT JOIN example
SELECT students.name, students.course, marks.marks
FROM students
LEFT JOIN marks
ON students.id = marks.student_id;

-- RIGHT JOIN
-- Get all marks records, even if there is no matching student
SELECT students.name, marks.marks
FROM students
RIGHT JOIN marks
ON students.id = marks.student_id;

-- Another RIGHT JOIN example
SELECT students.name, students.course, marks.marks
FROM students
RIGHT JOIN marks
ON students.id = marks.student_id;

-- FULL JOIN
-- Get all records from both tables
SELECT students.name, marks.marks
FROM students
FULL JOIN marks
ON students.id = marks.student_id;

-- Another FULL JOIN example
SELECT students.name, students.course, marks.marks
FROM students
FULL JOIN marks
ON students.id = marks.student_id;

-- SUBQUERY
-- A query written inside another query


-- SINGLE-ROW SUBQUERY
-- Find students who scored more than Rashid
SELECT name, marks
FROM students
WHERE marks > (
    SELECT marks
    FROM students
    WHERE name = 'Rashid'
);

-- Find the student with the highest marks
SELECT name, marks
FROM students
WHERE marks = (
    SELECT MAX(marks)
    FROM students
);


-- MULTIPLE-ROW SUBQUERY
-- Find students whose marks are greater than any
-- student taking Python
SELECT name, marks
FROM students
WHERE marks > ANY (
    SELECT marks
    FROM students
    WHERE course = 'Python'
);

-- Find students whose marks match any
-- marks of Python students
SELECT name, marks
FROM students
WHERE marks IN (
    SELECT marks
    FROM students
    WHERE course = 'Python'
);


-- NESTED SUBQUERY
-- Find the student with the highest marks
-- among students older than 18
SELECT name, marks
FROM students
WHERE marks = (
    SELECT MAX(marks)
    FROM students
    WHERE age > 18
);


-- SUBQUERY WITH IN
-- Find students who are enrolled in courses
-- that have students younger than 18
SELECT name, course
FROM students
WHERE course IN (
    SELECT course
    FROM students
    WHERE age < 18
);
